                localStorage.setItem('theme', 'light');
            } else {
                htmlTag.classList.add('dark');
                themeIcon.className = 'fa-solid fa-sun text-amber-400';
                localStorage.setItem('theme', 'dark');
            }
        }

        // Initialize Theme from Preferences
        (function initTheme() {
            const savedTheme = localStorage.getItem('theme');
            const themeIcon = document.getElementById('theme-icon');
            if (savedTheme === 'dark' || (!savedTheme && window.matchMedia('(prefers-color-scheme: dark)').matches)) {
                document.documentElement.classList.add('dark');
                if (themeIcon) themeIcon.className = 'fa-solid fa-sun text-amber-400';
            } else {
                document.documentElement.classList.remove('dark');
                if (themeIcon) themeIcon.className = 'fa-solid fa-moon';
            }
        })();

        // Language Switcher Toggle
        function toggleLanguage() {
            currentLang = currentLang === 'VI' ? 'EN' : 'VI';
            document.getElementById('current-lang').innerText = currentLang === 'VI' ? 'EN' : 'VI';
            
            const elements = document.querySelectorAll('[data-lang-vi]');
            elements.forEach(el => {
                const textVi = el.getAttribute('data-lang-vi');
                const textEn = el.getAttribute('data-lang-en');
                
                if (currentLang === 'EN' && textEn) {
                    el.textContent = textEn;
                } else if (currentLang === 'VI' && textVi) {
                    el.textContent = textVi;
                }
            });
        }

        // Mobile Menu Drawer Toggle
        function toggleMobileMenu() {
            const menu = document.getElementById('mobile-menu');
            menu.classList.toggle('hidden');
        }

        // Project Category Filter
        function filterProjects(category) {
            const cards = document.querySelectorAll('.project-card');
            const filterBtns = document.querySelectorAll('.project-filter-btn');

            // Update button styles
            filterBtns.forEach(btn => {
                if (btn.getAttribute('data-filter') === category) {
                    btn.className = 'project-filter-btn px-4 py-2 rounded-xl text-sm font-semibold transition bg-blue-600 text-white shadow-md shadow-blue-500/20';
                } else {
                    btn.className = 'project-filter-btn px-4 py-2 rounded-xl text-sm font-semibold transition bg-slate-100 dark:bg-slate-800 text-slate-700 dark:text-slate-300 hover:bg-slate-200 dark:hover:bg-slate-700';
                }
            });

            // Show/Hide Cards
            cards.forEach(card => {
                if (category === 'all' || card.classList.contains(category)) {
                    card.style.display = 'flex';
                } else {
                    card.style.display = 'none';
                }
            });
        }

        // Form Submit Simulation
        function handleFormSubmit(event) {
            event.preventDefault();
            const statusDiv = document.getElementById('form-status');
            statusDiv.classList.remove('hidden', 'bg-red-100', 'text-red-700', 'bg-green-100', 'text-green-700');
            
            statusDiv.classList.add('bg-green-100', 'text-green-700', 'dark:bg-green-950/60', 'dark:text-green-300');
            statusDiv.innerText = currentLang === 'VI' 
                ? 'Cảm ơn bạn! Lời nhắn đã được gửi thành công.' 
                : 'Thank you! Your message has been sent successfully.';
            
            document.getElementById('contact-form').reset();
            setTimeout(() => {
                statusDiv.classList.add('hidden');
            }, 5000);
        }
